\begin{center}
\section*{Inequalities.}
\end{center}

For positive $a$, $b$, $c$, $d$ prove the following inequalities

\medskip\medskip\medskip

\begin{enumerate}
 \item $a^{2}+b^{2}+c^{2}\geq ab+bc+ca$.

\item $a^{4}+b^{4}+c^{4}+d^{4}\geq 4abcd$.

\item $a(1+b)+b(1+c)+c(1+a)\geq 6\sqrt{abc}$.

\item $(a+b)(b+c)(a+c)\geq 8abc$.

\item $(1+ab)^{2}+(1+cd)^{2}+(ac)^{2}+(bd)^{2}\geq 1$.

\item Prove inequality $\frac{a^{2}}{4}+b^{2}+c^{2}\geq ab-ac+2bc$.

\item For any $x,y,z$ prove the following: $$x^{4}+y^{4}+z^{2}+1\geq x(xy^{2}-x+z+1).$$

\item For $n\geq3$ prove inequality: $$\frac{1}{n+1}+\frac{1}{n+2}+...+\frac{1}{2n}>\frac{3}{5}.$$

\item Which of the numbers $2^{2^{.^{.^{.^{2}}}}}$ (10 times 2) or $3^{3^{.^{.^{.^{3}}}}}$ (9 times 3) is larger?

\item For $n\geq1$ prove: $3^{n}>n\cdot 2^{n}$.

\item For $n\geq4$ prove: $2^{n}<n!$.

\item Let $a,b,c\geq0$. Prove: $ab+bc+ca\geq a\sqrt{bc}+b\sqrt{ac}+c\sqrt{ab}$.

\item Let $a,b,c$~--- be positive. Prove $\frac{bc}{a}+\frac{ac}{b}+\frac{ab}{c}\geq (a+b+c)$

\item $n$~--- natural number. Prove: $\sqrt{n+1}-\sqrt{n-1}>\frac{1}{\sqrt{n}}$.

\end{enumerate}
